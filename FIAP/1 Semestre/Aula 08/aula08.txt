Aula 08 — Estruturas de decisão: if, else e else if
Objetivo: permitir que o programa tome decisões conforme determinadas condições.


let nota = 7;

if (nota >= 7) {
  console.log("Aprovado");
} else {
  console.log("Reprovado");
}


2. Prática — Verificador de notas


function verificarNota(nota) {
  if (nota >= 7) {
    return "Aprovado";
  } else if (nota >= 5) {
    return "Recuperação";
  } else {
    return "Reprovado";
  }
}
